-- Basit bağlantı yöneticisi (merkezi temizlik için)
local Conns = {}
local function track(key, conn)
	if Conns[key] ~= nil then
		pcall(function() Conns[key]:Disconnect() end)
	end
	Conns[key] = conn
	return conn
end
local function untrackAll()
	for k, c in pairs(Conns) do
		pcall(function() c:Disconnect() end)
		Conns[k] = nil
	end
end

-- Orijinalleri sakla (restore için)
local Originals = {}
do
	local char = LocalPlayer and LocalPlayer.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	if hum ~= nil then
		Originals.WalkSpeed = hum.WalkSpeed
		Originals.JumpPower = hum.JumpPower
	end
	local cam = Workspace.CurrentCamera
	if cam ~= nil then
		Originals.FOV = cam.FieldOfView
	end
end
if LocalPlayer ~= nil then
	track("CharAdded", LocalPlayer.CharacterAdded:Connect(function()
		task.defer(function()
			task.wait(0.5)
			local char = LocalPlayer.Character
			local hum = char and char:FindFirstChildOfClass("Humanoid")
			if hum ~= nil then
				if Originals.WalkSpeed == nil then Originals.WalkSpeed = hum.WalkSpeed end
				if Originals.JumpPower == nil then Originals.JumpPower = hum.JumpPower end
			end
		end)
	end))
end

-- Hedef önceliği
local TargetPriority = "Crosshair"
local LockedTarget = nil

local function getLocalRoot()
	local char = LocalPlayer and LocalPlayer.Character
	local root = char and char:FindFirstChild("HumanoidRootPart")
	if root ~= nil and root:IsA("BasePart") then
		return root
	end
	return nil
end

local function findPlayers(query)
	local out = {}
	if type(query) ~= "string" or #query < 1 then
		return out
	end
	local needle = string.lower(query)
	for _, p in ipairs(Players:GetPlayers()) do
		local name = ""
		local disp = ""
		pcall(function() name = p.Name end)
		pcall(function() disp = p.DisplayName end)
		if string.find(string.lower(name .. " " .. disp), needle, 1, true) ~= nil then
			table.insert(out, p)
		end
	end
	return out
end

local function hasLineOfSight(fromPos, toPos, targetModel)
	local dir = toPos - fromPos
	if dir.Magnitude < 0.001 then
		return true
	end
	local ignore = {}
	if LocalPlayer ~= nil and LocalPlayer.Character ~= nil then
		table.insert(ignore, LocalPlayer.Character)
	end
	if targetModel ~= nil then
		table.insert(ignore, targetModel)
	end
	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	params.IgnoreWater = true
	if #ignore > 0 then
		params.FilterDescendantsInstances = ignore
	end
	local ok, result = pcall(function()
		return Workspace:Raycast(fromPos, dir, params)
	end)
	if not ok then
		return false
	end
	if result == nil then
		return true
	end
	return (result.Position - toPos).Magnitude <= 2
end

local function isValidTarget(player)
	if player == nil or player.Parent == nil then
		return false
	end
	if player == LocalPlayer then
		return false
	end
	local char = player.Character
	if char == nil then
		return false
	end
	local hum = char:FindFirstChildOfClass("Humanoid")
	if hum == nil then
		return false
	end
	local hpOk, hp = pcall(function() return hum.Health end)
	if not hpOk or hp <= 0 then
		return false
	end
	-- Takım kontrolü
	local sameTeam = false
	pcall(function()
		sameTeam = (LocalPlayer.Team ~= nil and LocalPlayer.Team == player.Team)
	end)
	if sameTeam then
		return false
	end
	-- Mesafe (500)
	local lroot = getLocalRoot()
	local troot = char:FindFirstChild("HumanoidRootPart")
	if lroot ~= nil and troot ~= nil and troot:IsA("BasePart") then
		if (lroot.Position - troot.Position).Magnitude > 500 then
			return false
		end
	end
	-- Görünürlük
	local cam = Workspace.CurrentCamera
	if cam ~= nil and troot ~= nil and troot:IsA("BasePart") then
		if not hasLineOfSight(cam.CFrame.Position, troot.Position, char) then
			return false
		end
	end
	return true
end

local function screenDist(player)
	local cam = Workspace.CurrentCamera
	if cam == nil then
		return nil
	end
	local char = player.Character
	local part = char and char:FindFirstChild("HumanoidRootPart")
	if part == nil or not part:IsA("BasePart") then
		return nil
	end
	local ok, vec, onScreen = pcall(function()
		return cam:WorldToViewportPoint(part.Position)
	end)
	if not ok or onScreen == false then
		return nil
	end
	local okM, mouse = pcall(function()
		return UserInputService:GetMouseLocation()
	end)
	if not okM then
		return nil
	end
	return (Vector2.new(vec.X, vec.Y) - mouse).Magnitude
end

local function getBestTarget()
	local best = nil
	if TargetPriority == "Distance" then
		local bestD = math.huge
		local lroot = getLocalRoot()
		for _, p in ipairs(Players:GetPlayers()) do
			if isValidTarget(p) then
				local char = p.Character
				local troot = char and char:FindFirstChild("HumanoidRootPart")
				if lroot ~= nil and troot ~= nil and troot:IsA("BasePart") then
					local d = (lroot.Position - troot.Position).Magnitude
					if d < bestD then
						bestD = d
						best = p
					end
				end
			end
		end
	elseif TargetPriority == "LowestHealth" then
		local bestH = math.huge
		for _, p in ipairs(Players:GetPlayers()) do
			if isValidTarget(p) then
				local hum = p.Character and p.Character:FindFirstChildOfClass("Humanoid")
				if hum ~= nil then
					local ok, hp = pcall(function() return hum.Health end)
					if ok and hp < bestH then
						bestH = hp
						best = p
					end
				end
			end
		end
	else
		local bestS = 120
		for _, p in ipairs(Players:GetPlayers()) do
			if isValidTarget(p) then
				local sd = screenDist(p)
				if sd ~= nil and sd <= bestS then
					bestS = sd
					best = p
				end
			end
		end
	end
	return best
end

local function doPanic(windowRef)
	LockedTarget = nil
	ESPOn = false
	FlyOn = false
	AimOn = false
	TrigOn = false
	RepOn = false
	WSOn = false
	InfJumpOn = false
	if ESPFolder ~= nil then
		pcall(function() ESPFolder:Destroy() end)
		ESPFolder = nil
	end
	if XHairGui ~= nil then
		pcall(function() XHairGui:Destroy() end)
		XHairGui = nil
		XHairParts = {}
	end
	local char = LocalPlayer and LocalPlayer.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	if hum ~= nil then
		if Originals.WalkSpeed ~= nil then
			pcall(function() hum.WalkSpeed = Originals.WalkSpeed end)
		end
		if Originals.JumpPower ~= nil then
			pcall(function() hum.JumpPower = Originals.JumpPower end)
		end
	end
	untrackAll()
	if windowRef ~= nil then
		pcall(function()
			windowRef:Notify({ Title = "Panic", Description = "Disabled, values restored", Lifetime = 3 })
		end)
	end
	-- Panic tuşunu geri bağla
	track("PanicKey", UserInputService.InputBegan:Connect(function(input, gpe)
		if not gpe and input.KeyCode == Enum.KeyCode.F8 then
			doPanic(windowRef)
		end
	end))
end

	return loadstring(src)()

-- ================= PHASE 3: ESP + Crosshair =================
ESPOn = false
ESPMaxDist = 500
ESPTeamColors = true
ESPColor = Color3.fromRGB(255, 255, 255)
ESPFolder = nil

local function espRebuild()
	if ESPFolder ~= nil then
		pcall(function() ESPFolder:Destroy() end)
		ESPFolder = nil
	end
	if not ESPOn then
		return
	end
	local folder = Instance.new("Folder")
	folder.Name = "UTK_ESP"
	folder.Parent = Workspace
	ESPFolder = folder
	local lr = getLocalRoot()
	for _, p in ipairs(Players:GetPlayers()) do
		if p ~= LocalPlayer then
			local char = p.Character
			local hum = char and char:FindFirstChildOfClass("Humanoid")
			if hum ~= nil then
				local hpOk, hp = pcall(function() return hum.Health end)
				if hpOk and hp > 0 then
					local hrp = char:FindFirstChild("HumanoidRootPart")
					local dist = 0
					if lr ~= nil and hrp ~= nil and hrp:IsA("BasePart") then
						dist = (lr.Position - hrp.Position).Magnitude
					end
					if dist <= ESPMaxDist then
						local color = ESPColor
						if ESPTeamColors then
							pcall(function() color = p.TeamColor.Color end)
						end
						local hl = Instance.new("Highlight")
						hl.Adornee = char
						hl.FillColor = color
						hl.FillTransparency = 0.6
						hl.OutlineColor = color
						hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
						hl.Parent = folder
						local head = char:FindFirstChild("Head")
						if head ~= nil and head:IsA("BasePart") then
							local bb = Instance.new("BillboardGui")
							bb.Adornee = head
							bb.Size = UDim2.fromOffset(120, 32)
							bb.StudsOffset = Vector3.new(0, 1.5, 0)
							bb.AlwaysOnTop = true
							local tl = Instance.new("TextLabel")
							tl.BackgroundTransparency = 1
							tl.Size = UDim2.fromScale(1, 1)
							tl.Font = Enum.Font.GothamBold
							tl.TextSize = 12
							tl.TextStrokeTransparency = 0.5
							tl.TextColor3 = color
							tl.Text = p.DisplayName .. " (" .. math.floor(dist) .. "m)"
							tl.Parent = bb
							bb.Parent = folder
						end
					end
				end
			end
		end
	end
end

XHairOn = false
XHairGui = nil
XHairParts = {}
XHairSize = 12
XHairGap = 6
XHairColor = Color3.fromRGB(0, 255, 0)

local function xhairRefresh()
	if XHairParts.Top == nil then
		return
	end
	XHairParts.Top.Size = UDim2.fromOffset(2, XHairSize)
	XHairParts.Top.Position = UDim2.new(0.5, 0, 0.5, -(XHairGap + XHairSize / 2))
	XHairParts.Bottom.Size = UDim2.fromOffset(2, XHairSize)
	XHairParts.Bottom.Position = UDim2.new(0.5, 0, 0.5, (XHairGap + XHairSize / 2))
	XHairParts.Left.Size = UDim2.fromOffset(XHairSize, 2)
	XHairParts.Left.Position = UDim2.new(0.5, -(XHairGap + XHairSize / 2), 0.5, 0)
	XHairParts.Right.Size = UDim2.fromOffset(XHairSize, 2)
	XHairParts.Right.Position = UDim2.new(0.5, (XHairGap + XHairSize / 2), 0.5, 0)
	XHairParts.Dot.Size = UDim2.fromOffset(2, 2)
	XHairParts.Dot.Position = UDim2.new(0.5, 0, 0.5, 0)
	for _, f in pairs(XHairParts) do
		f.BackgroundColor3 = XHairColor
	end
end

local function xhairSet(on)
	XHairOn = on
	if XHairGui ~= nil then
		pcall(function() XHairGui:Destroy() end)
		XHairGui = nil
		XHairParts = {}
	end
	if not on then
		return
	end
	local parent = LocalPlayer and LocalPlayer:FindFirstChildOfClass("PlayerGui")
	if parent == nil then
		return
	end
	local gui = Instance.new("ScreenGui")
	gui.Name = "UTK_Crosshair"
	gui.ResetOnSpawn = false
	gui.IgnoreGuiInset = true
	gui.DisplayOrder = 9999
	for _, n in ipairs({ "Top", "Bottom", "Left", "Right", "Dot" }) do
		local f = Instance.new("Frame")
		f.Name = n
		f.BorderSizePixel = 0
		f.AnchorPoint = Vector2.new(0.5, 0.5)
		f.Parent = gui
		XHairParts[n] = f
	end
	xhairRefresh()
	gui.Parent = parent
	XHairGui = gui
end

-- ================= PHASE 4: Movement + Camera =================
WSOn = false
WSVal = 16
OrigWS = OrigWS or Originals.WalkSpeed
OrigJP = OrigJP or Originals.JumpPower
FlyOn = false
FlySpeed = 50
InfJumpOn = false

local function applyWalkSpeed()
	local char = LocalPlayer and LocalPlayer.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	if hum == nil then
		return
	end
	if WSOn then
		pcall(function() hum.WalkSpeed = WSVal end)
	elseif OrigWS ~= nil then
		pcall(function() hum.WalkSpeed = OrigWS end)
	end
end

track("InfJump", UserInputService.JumpRequest:Connect(function()
	if InfJumpOn then
		local char = LocalPlayer and LocalPlayer.Character
		local hum = char and char:FindFirstChildOfClass("Humanoid")
		if hum ~= nil then
			pcall(function() hum:ChangeState(Enum.HumanoidStateType.Jumping) end)
		end
	end
end))

if LocalPlayer ~= nil then
	track("WSRespawn", LocalPlayer.CharacterAdded:Connect(function()
		task.defer(function()
			task.wait(0.5)
			applyWalkSpeed()
		end)
	end))
end

do
	local m = makeSection("Movement", "Left")
	if m ~= nil then
		m:Toggle({ Name = "WalkSpeed Override", Default = false, Callback = function(on)
			WSOn = on
			applyWalkSpeed()
		end }, "WS_Toggle")
		m:Slider({ Name = "WalkSpeed", Default = 16, Minimum = 16, Maximum = 200, DisplayMethod = "Value", Precision = 0, Callback = function(val)
			WSVal = val
			applyWalkSpeed()
		end }, "WS_Slider")
		m:Slider({ Name = "JumpPower", Default = 50, Minimum = 0, Maximum = 300, DisplayMethod = "Value", Precision = 0, Callback = function(val)
			local char = LocalPlayer and LocalPlayer.Character
			local hum = char and char:FindFirstChildOfClass("Humanoid")
			if hum ~= nil then
				pcall(function() hum.JumpPower = val end)
			end
		end }, "JP_Slider")
		m:Toggle({ Name = "Fly (WASD+Space/Shift)", Default = false, Callback = function(on)
			FlyOn = on
			if on then
				local RunService = game:GetService("RunService")
				track("FlyLoop", RunService.Heartbeat:Connect(function(dt)
					if not FlyOn then
						return
					end
					local char = LocalPlayer and LocalPlayer.Character
					local root = char and char:FindFirstChild("HumanoidRootPart")
					local cam = Workspace.CurrentCamera
					if root == nil or cam == nil or not root:IsA("BasePart") then
						return
					end
					local move = Vector3.new(0, 0, 0)
					if UserInputService:IsKeyDown(Enum.KeyCode.W) then move = move + cam.CFrame.LookVector end
					if UserInputService:IsKeyDown(Enum.KeyCode.S) then move = move - cam.CFrame.LookVector end
					if UserInputService:IsKeyDown(Enum.KeyCode.A) then move = move - cam.CFrame.RightVector end
					if UserInputService:IsKeyDown(Enum.KeyCode.D) then move = move + cam.CFrame.RightVector end
					if UserInputService:IsKeyDown(Enum.KeyCode.Space) then move = move + Vector3.new(0, 1, 0) end
					if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then move = move - Vector3.new(0, 1, 0) end
					if move.Magnitude > 0 then
						root.CFrame = root.CFrame + (move.Unit * FlySpeed * dt)
					end
					root.Velocity = Vector3.new(0, 0, 0)
				end))
			else
				if Conns["FlyLoop"] ~= nil then
					pcall(function() Conns["FlyLoop"]:Disconnect() end)
					Conns["FlyLoop"] = nil
				end
			end
		end }, "Fly_Toggle")
		m:Slider({ Name = "Fly Speed", Default = 50, Minimum = 10, Maximum = 300, DisplayMethod = "Value", Precision = 0, Callback = function(val)
			FlySpeed = val
		end }, "Fly_Speed")
		m:Toggle({ Name = "Infinite Jump", Default = false, Callback = function(on)

-- ================= PHASE 5: Aim Assist + Trigger =================
AimOn = false
AimSmooth = 0.2
AimFOV = 120
TrigOn = false
TrigDelay = 0.15
TrigCD = 0.5
TrigLast = 0

do
	local c = makeSection("Combat", "Left")
	if c ~= nil then
		c:Toggle({ Name = "Aim Assist (hold RMB)", Default = false, Callback = function(on)
			AimOn = on
			if on then
				local RunService = game:GetService("RunService")
				track("AimLoop", RunService.RenderStepped:Connect(function()
					if not AimOn then
						return
					end
					local holding = false
					pcall(function()
						holding = UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2)
					end)
					if not holding then
						return
					end
					local cam = Workspace.CurrentCamera
					if cam == nil then
						return
					end
					local best, bestD = nil, AimFOV
					local okM, mouse = pcall(function() return UserInputService:GetMouseLocation() end)
					if not okM then
						return
					end
					for _, p in ipairs(Players:GetPlayers()) do
						if isValidTarget(p) then
							local char = p.Character
							local head = char and (char:FindFirstChild("Head") or char:FindFirstChild("HumanoidRootPart"))
							if head ~= nil and head:IsA("BasePart") then
								local ok, vec, onScr = pcall(function() return cam:WorldToViewportPoint(head.Position) end)
								if ok and onScr then
									local d = (Vector2.new(vec.X, vec.Y) - mouse).Magnitude
									if d < bestD then
										bestD = d
										best = head
									end
								end
							end
						end
					end
					if best ~= nil then
						local goal = CFrame.new(cam.CFrame.Position, best.Position)
						cam.CFrame = cam.CFrame:Lerp(goal, math.clamp(AimSmooth, 0.01, 1))
					end
				end))
			else
				if Conns["AimLoop"] ~= nil then
					pcall(function() Conns["AimLoop"]:Disconnect() end)
					Conns["AimLoop"] = nil
				end
			end
		end }, "Aim_Toggle")
		c:Slider({ Name = "Aim Smoothness", Default = 20, Minimum = 1, Maximum = 100, DisplayMethod = "Percent", Precision = 0, Callback = function(val)
			AimSmooth = val / 100
		end }, "Aim_Smooth")
		c:Slider({ Name = "Aim FOV", Default = 120, Minimum = 30, Maximum = 400, DisplayMethod = "Value", Precision = 0, Callback = function(val)
			AimFOV = val
		end }, "Aim_FOV")
		c:Toggle({ Name = "Trigger Bot", Default = false, Callback = function(on)
			if on and type(mouse1click) ~= "function" then
				Window:Notify({ Title = "Trigger", Description = "Unavailable (mouse1click missing)", Lifetime = 3 })
				return
			end
			TrigOn = on
			if on then
				task.spawn(function()
					while TrigOn do
						task.wait(TrigDelay)
						if not TrigOn then
							break
						end
						if os.clock() - TrigLast >= TrigCD then
							local cam = Workspace.CurrentCamera
							local okM, mouse = pcall(function() return UserInputService:GetMouseLocation() end)
							if cam ~= nil and okM then
								for _, p in ipairs(Players:GetPlayers()) do
									if isValidTarget(p) then
										local char = p.Character
										local head = char and (char:FindFirstChild("Head") or char:FindFirstChild("HumanoidRootPart"))
										if head ~= nil and head:IsA("BasePart") then
											local ok, vec, onScr = pcall(function() return cam:WorldToViewportPoint(head.Position) end)
											if ok and onScr and (Vector2.new(vec.X, vec.Y) - mouse).Magnitude <= 60 then
												TrigLast = os.clock()
												pcall(function() mouse1click() end)
												break
											end
										end
									end
								end
							end
						end
					end
				end)
			end
		end }, "Trig_Toggle")
	end
end
